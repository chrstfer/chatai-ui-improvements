// src/content/controllers/DomObserver.ts
import { processCodeBlock } from '../models/CodeBlockParser';
import { injectFooterActions } from '../views/FooterActions';

export class DomObserver {
    private observer: MutationObserver;

    constructor() {
        this.observer = new MutationObserver(this.handleMutations.bind(this));
    }

    public observe(target: HTMLElement): void {
        this.observer.observe(target, { childList: true, subtree: true });
    }

    private handleMutations(mutations: MutationRecord[]): void {
        for (const mutation of mutations) {
            if (mutation.type !== 'childList') continue;

            mutation.addedNodes.forEach((node) => {
                if (node.nodeType !== Node.ELEMENT_NODE) return;
                const el = node as HTMLElement;

                // 1. Intercept newly rendered code blocks
                if (el.tagName.toLowerCase() === 'code-block') {
                    processCodeBlock(el);
                } else {
                    const codeBlocks = el.querySelectorAll('code-block');
                    codeBlocks.forEach(processCodeBlock);
                }

                // 2. Intercept response footers for action injection
                if (el.matches('message-actions[footer]')) {
                    injectFooterActions(el);
                } else {
                    const footers = el.querySelectorAll('message-actions[footer]');
                    footers.forEach(injectFooterActions);
                }
            });
        }
    }
}

    if (parsed) {

function parseCodeBlockData(node: HTMLElement): ParsedCodeBlock | null {
    const firstChild = node.firstChild;
    let language = 'unknown';

    // Verify first child is a text node to extract language metadata
    if (firstChild && firstChild.nodeType === Node.TEXT_NODE) {
        const text = firstChild.textContent?.trim() || '';
        if (text === 'Plain text' || text === 'Code snippet') {
            language = 'text';
        } else if (text) {
            language = text.toLowerCase();
        }
    }

    const codeElement = node.querySelector('code');
    if (!codeElement) return null;

    // Extracting raw text. Using textContent safely ignores HTML tags 
    // injected by server-side or client-side syntax highlighters.
    const rawContent = codeElement.textContent || '';

    return { language, rawContent, node };
}

#+BEGIN_SRC typescript
import { CodeBlockParser } from '../model/CodeBlockParser';
import { DocumentView } from '../view/DocumentView';
import { FooterActions } from '../view/FooterActions';

export class GeminiObserver {
  private observer: MutationObserver | null = null;

  public start(root: HTMLElement = document.body): void {
    this.stop();
    this.observer = new MutationObserver(this.handleMutations.bind(this));
    this.observer.observe(root, {
      childList: true,
      subtree: true
    });

    // Run an initial sweep for pre-existing nodes
    this.scanSubtree(root);
  }

  public stop(): void {
    if (this.observer) {
      this.observer.disconnect();
      this.observer = null;
    }
  }

  private handleMutations(mutations: MutationRecord[]): void {
    for (let i = 0; i < mutations.length; i++) {
      const mutation = mutations[i];
      if (mutation.type !== 'childList') continue;

      mutation.addedNodes.forEach((node) => {
        if (node.nodeType === Node.ELEMENT_NODE) {
          this.scanSubtree(node as HTMLElement);
        }
      });
    }
  }

  private scanSubtree(element: HTMLElement): void {
    // 1. Process code-block elements
    if (element.tagName.toLowerCase() === 'code-block') {
      this.handleCodeBlock(element);
    } else {
      const codeBlocks = element.querySelectorAll<HTMLElement>('code-block');
      codeBlocks.forEach((cb) => this.handleCodeBlock(cb));
    }

    // 2. Process footer elements
    if (element.matches('message-actions[footer]')) {
      this.handleFooter(element);
    } else {
      const footers = element.querySelectorAll<HTMLElement>('message-actions[footer]');
      footers.forEach((f) => this.handleFooter(f));
    }
  }

  private handleCodeBlock(codeBlockNode: HTMLElement): void {
    const parsed = CodeBlockParser.parse(codeBlockNode);
    if (!parsed) return;

    // Filter by relevant languages before triggering view transformation
    if (parsed.language === 'org' || parsed.language === 'text') {
      DocumentView.mount(parsed);
    }
  }

  private handleFooter(footerNode: HTMLElement): void {
    FooterActions.inject(footerNode, () => {
      const responseRoot = footerNode.closest<HTMLElement>('model-response');
      if (!responseRoot) return;

      const markdownPanel = responseRoot.querySelector<HTMLElement>('.markdown-main-panel');
      if (!markdownPanel) return;

      // Extract payload and write to clipboard
      const content = markdownPanel.innerText;
      navigator.clipboard.writeText(content).catch((err) => {
        console.error('Failed to copy org content:', err);
      });
    });
  }
}

**** Preact 11 Integration & MutationObserver Topology <2026-09-05 Sat 23:16>

***** 1. Architectural Decision: Single Central Observer vs. Multiple Targeted Observers

A *single central ~MutationObserver~* is superior to multiple targeted observers for SPA content scripts.

****** Mechanical Rationale:
1. **Observer Heap & Microtask Queue Overhead:** Each active ~MutationObserver~
   allocates its own callback dispatch queue. In dynamic single-page
   applications undergoing frequent token-streaming DOM mutations, multiple
   observers generate redundant queue flushes and overlapping subtree scans.
2. **Dynamic Subtree Desynchronization:** Targeted observers bound to ephemeral
   DOM nodes (e.g., ~infinite-scroller~ or specific conversation threads) are
   prone to detachment during client-side route changes or internal SPA
   re-renders. A single global observer attached to an invariant root (e.g.,
   ~document.documentElement~ or ~document.body~) survives all internal client
   router cycles.
3. **Batching & Unified Traversal:** A central observer receives an aggregated
   batch of mutations, allowing for an efficient batch pass over added/removed
   nodes and feeding site-specific registry listeners.

The idiomatic architecture uses a **Central Observer Bus** that routes added/removed nodes to site-specific adapter callbacks.

---

***** 2. Preact 11 Component Injection & Portal Lifecycle

In Preact 11, mounting components into foreign DOM elements requires proper lifecycle bookkeeping. Calling ~render(h(Component, props), containerNode)~ attaches a Virtual DOM tree. When the SPA destroys or recycles that DOM node, or when the extension unmounts, you must explicitly call ~render(null, containerNode)~ to trigger cleanup effects and prevent memory leaks.

import { render } from 'preact';

/**
 * Manages mounted Preact roots to guarantee clean teardown and avoid detached VNode leaks.
 */
export class MountRegistry {
  private static mountedContainers = new Set<HTMLElement>();

  public static register(container: HTMLElement): void {
    this.mountedContainers.add(container);
  }

  public static unmount(container: HTMLElement): void {
    if (this.mountedContainers.has(container)) {
      render(null, container);
      this.mountedContainers.delete(container);
    }
  }

  public static unmountAll(): void {
    for (const container of this.mountedContainers) {
      render(null, container);
    }
    this.mountedContainers.clear();
  }
}

import { h, render, ComponentType } from 'preact';
import { ExtractedCodeBlock, DocumentViewProps } from '../types';
import { MountRegistry } from './MountRegistry';

export class DocumentViewHost {
  public static mount(
    block: ExtractedCodeBlock,
    Component: ComponentType<DocumentViewProps>
  ): void {
    const { container } = block;

    // Prevent duplicate mounting
    if (container.dataset.preactMounted === 'true') {
      return;
    }

    // Isolate native code rendering without breaking Gemini's layout tree
    const hostMountPoint = document.createElement('div');
    hostMountPoint.className = 'gemini-extension-document-view-host';

    // Hide original children while preserving them in the DOM if needed for fallback
    Array.from(container.children).forEach((child) => {
      (child as HTMLElement).style.display = 'none';
    });

    container.appendChild(hostMountPoint);
    container.dataset.preactMounted = 'true';

    const props: DocumentViewProps = {
      content: block.rawContent,
      language: block.language,
      hasNativeHeader: block.hasNativeHeader,
      rawContainer: container
    };

    // Preact 11 render call
    render(h(Component, props), hostMountPoint);
    MountRegistry.register(hostMountPoint);
  }

  public static cleanup(container: HTMLElement): void {
    const host = container.querySelector<HTMLElement>('.gemini-extension-document-view-host');
    if (host) {
      MountRegistry.unmount(host);
      host.remove();
    }
    delete container.dataset.preactMounted;
  }
}

***** Observer Topology: Centralized Pipeline vs. Isolated Site Observers

Using separate, uncoordinated `MutationObserver` instances per site within the
same runtime environment is an anti-pattern. However, hardcoding site-specific
selector logic inside a single monolithic observer violates open-closed
architecture.

The optimal pattern is a **Single Central Engine with a Strategy-Based Site
Adapter Pipeline**:

#+BEGIN_SRC text

                       [ Mutation Dispatcher / Core ]
                                       │
                      ┌────────────────┴────────────────┐
                      ▼                                 ▼
          [ Active Site Strategy ]           [ (Inactive Strategies) ]
             (e.g., GeminiAdapter)              (Claude, ChatGPT, etc.)

   minimizes overhead.
2. **Dynamic Route Awareness:** Chat SPAs continuously mutate their subtrees. A centralized orchestrator activates only the site strategy matching `window.location.hostname`.
3. **Decoupled Lifecycle:** The strategy interface accepts raw added nodes and decides whether to ignore, inspect, or process them.

***** Preact 11 Mounting Semantics in Web Extension Content Scripts

Preact 11 modernizes its rendering pipeline (notably removing legacy signatures like the 3-parameter `replaceNode` variant from `render()`). When injecting Preact components into an external host DOM:

1. **Shadow DOM Encapsulation:** Injecting directly into the host DOM risks CSS collisions from the host app's styling sheets. Creating an open or closed `ShadowRoot` provides style containment.
2. **Controlled Unmounting / Teardown:** Host SPAs frequently recycle DOM elements during virtual scrolling (e.g., `infinite-scroller`). When a parent container is removed, you must call `render(null, mountPoint)` to prevent memory leaks and unbind Preact event listeners.

export class CodeBlockParser {
  public static parse(node: HTMLElement): ExtractedCodeBlock | null {
    const codeElement = node.querySelector<HTMLElement>('code');
    if (!codeElement) return null;

    let language: SupportedLanguage = 'unknown';
    let hasNativeHeader = false;

    const firstChild = node.firstChild;
    if (firstChild && firstChild.nodeType === Node.TEXT_NODE) {
      hasNativeHeader = true;
      const rawLabel = (firstChild.textContent ?? '').trim();
      
      if (rawLabel === 'Plain text' || rawLabel === 'Code snippet') {
        language = 'text';
      } else if (rawLabel.length > 0) {
        language = rawLabel.toLowerCase();
      }
    }

    // Direct textContent extraction cleanly bypasses client-side syntax highlight spans
    const rawContent = codeElement.textContent ?? '';

    return {
      language,
      rawContent,
      container: node,
      codeElement,
      hasNativeHeader
    };
  }
}

#+BEGIN_SRC typescript
import { CodeBlockParser } from '../model/CodeBlockParser';
import { DocumentMount } from '../view/DocumentMount';

export class GeminiObserver {
  private observer: MutationObserver | null = null;

  public attach(root: HTMLElement): void {
    this.detach();

    this.observer = new MutationObserver(this.handleMutations.bind(this));
    this.observer.observe(root, {
      childList: true,
      subtree: true
    });

    // Initial pass over any server-rendered or already-loaded messages
    this.scanSubtree(root);
  }

  public detach(): void {
    if (this.observer) {
      this.observer.disconnect();
      this.observer = null;
    }
  }

  private handleMutations(mutations: MutationRecord[]): void {
    for (let i = 0; i < mutations.length; i++) {
      const mutation = mutations[i];
      
      // Handle additions
      if (mutation.addedNodes.length > 0) {
        mutation.addedNodes.forEach((node) => {
          if (node.nodeType === Node.ELEMENT_NODE) {
            this.scanSubtree(node as HTMLElement);
          }
        });
      }

      // Handle removals to prevent memory leaks from virtualized scrolling
      if (mutation.removedNodes.length > 0) {
        mutation.removedNodes.forEach((node) => {
          if (node.nodeType === Node.ELEMENT_NODE) {
            this.handleRemovals(node as HTMLElement);
          }
        });
      }
    }
  }

  private scanSubtree(element: HTMLElement): void {
    if (element.tagName.toLowerCase() === 'code-block') {
      this.processCodeBlock(element);
      return;
    }

    const codeBlocks = element.querySelectorAll<HTMLElement>('code-block');
    for (let i = 0; i < codeBlocks.length; i++) {
      this.processCodeBlock(codeBlocks[i]);
    }
  }

  private handleRemovals(element: HTMLElement): void {
    if (element.tagName.toLowerCase() === 'code-block') {
      DocumentMount.unmount(element);
      return;
    }

    const codeBlocks = element.querySelectorAll<HTMLElement>('code-block');
    for (let i = 0; i < codeBlocks.length; i++) {
      DocumentMount.unmount(codeBlocks[i]);
    }
  }

  private processCodeBlock(node: HTMLElement): void {
    const parsed = CodeBlockParser.parse(node);
    if (!parsed) return;

    if (parsed.language === 'org' || parsed.language === 'text') {
      DocumentMount.mount(parsed);
    }
  }
}

****** Understanding Angular Production DOM Architecture
To target elements reliably in an Angular application without access to internal
models:
******* 1. Component Boundaries and Custom Tags
- Unlike generic React builds that compile mostly to ~<div>~soup, Angular
  components compile into their custom tag names by default (e.g.,
  ~<app-chat-message>~, ~<chat-list>~, ~<message-item>~).
- These custom element tags are the most stable CSS selectors available. Always
  prioritize custom tag names over classes.

******* 2. Scoped Encapsulation (~_ngcontent~ / ~_nghost~)
- Angular uses attribute selectors (like ~_ngcontent-ng-c123456789~) to scope
  CSS to specific components.
- *Never target these attributes directly in your selectors.* The hash suffix
  can and will change on every deployment or build artifact refresh.

******* 3. CDKs and Material Layouts
- If the application uses the Angular Component Dev Kit (CDK), the chat
  container likely uses virtual scrolling (~<cdk-virtual-scroll-viewport>~).
- *Virtual Scrolling Implication:* Only messages currently in the viewport exist
  in the DOM. As the user scrolls or new messages arrive, DOM nodes are
  destroyed and recycled. Your extension cannot rely on selecting elements once
  and caching them indefinitely.
****** Dynamic Streaming Chat Interception Strategy
To interact with streaming messages in an extension, polling is inefficient and
fragile. You need a reactive pipeline using native browser APIs within your
Content Script.

"""
DOM Workspace: Utilities for sanitizing, unwrapping, and visualizing
obfuscated or deeply nested Angular/modern web application DOM trees.
"""

import re
import sys
from pathlib import Path
from typing import Iterable, List, Optional, Set
from bs4 import BeautifulSoup, Comment, NavigableString, Tag


def load_soup(source: str, is_file: bool = True) -> BeautifulSoup:
    """Load and parse HTML from a file path or raw string using lxml."""
    if is_file:
        with open(source, "r", encoding="utf-8") as f:
            return BeautifulSoup(f, "lxml")
    return BeautifulSoup(source, "lxml")


def strip_comments(soup: BeautifulSoup) -> BeautifulSoup:
    """Remove all HTML comments (*ngIf, *ngFor, dynamic templates)."""
    for comment in soup.find_all(text=lambda t: isinstance(t, Comment)):
        comment.extract()
    return soup


def strip_tags(
    soup: BeautifulSoup,
    tags_to_drop: Iterable[str] = ("script", "style", "noscript", "svg", "link", "meta"),
) -> BeautifulSoup:
    """Decompose non-structural or visually heavy elements completely."""
    for tag in soup.find_all(list(tags_to_drop)):
        tag.decompose()
    return soup


def strip_ng_attributes(
    soup: BeautifulSoup,
    pattern: str = r"^_ng(content|host)-.*",
) -> BeautifulSoup:
    """Remove transient Angular build/scoping hash attributes."""
    compiled_pattern = re.compile(pattern)
    for element in soup.find_all(True):
        for attr in list(element.attrs.keys()):
            if compiled_pattern.match(attr):
                del element.attrs[attr]
    return soup


def unwrap_named_tags(
    soup: BeautifulSoup,
    tags_to_unwrap: Iterable[str] = ("ng-container", "ng-template"),
) -> BeautifulSoup:
    """Unwrap specific container tags while retaining their child nodes."""
    for tag in soup.find_all(list(tags_to_unwrap)):
        tag.unwrap()
    return soup


def unwrap_empty_containers(
    soup: BeautifulSoup,
    container_tags: Iterable[str] = ("div", "span"),
    preserve_classes: bool = False,
) -> BeautifulSoup:
    """
    Iteratively hoist child elements out of non-semantic wrapper tags
    that have no attributes (or no classes if preserve_classes=False).
    """
    flattened = True
    tags = list(container_tags)
    while flattened:
        flattened = False
        for node in soup.find_all(tags):
            # Check if container is bare
            if not node.attrs or (not preserve_classes and not node.get("id") and not node.get("class")):
                if len(node.contents) > 0:
                    node.unwrap()
                    flattened = True
    return soup


def sanitize_angular_dom(
    soup: BeautifulSoup,
    drop_tags: Iterable[str] = ("script", "style", "noscript", "svg"),
    unwrap_tags: Iterable[str] = ("ng-container", "ng-template"),
    flatten_bare_divs: bool = True,
) -> BeautifulSoup:
    """Convenience pipeline running standard cleanup passes in order."""
    strip_comments(soup)
    strip_tags(soup, drop_tags)
    strip_ng_attributes(soup)
    unwrap_named_tags(soup, unwrap_tags)
    if flatten_bare_divs:
        unwrap_empty_containers(soup)
    return soup


def render_dom_tree(
    node: Tag,
    prefix: str = "",
    is_last: bool = True,
    max_depth: int = 15,
    current_depth: int = 0,
    show_attrs: bool = True,
    max_text_len: int = 40,
) -> List[str]:
    """
    Recursively renders a node into standard Unix `tree` styled output.
    Displays tag names, IDs, classes, and truncated text content.
    """
    if current_depth > max_depth or not isinstance(node, Tag):
        return []

    lines = []
    
    # Format current node summary
    connector = "└── " if is_last else "├── "
    label = [f"<{node.name}>"]

    if show_attrs:
        if node.get("id"):
            label.append(f"#{node['id']}")
        if node.get("class"):
            classes = ".".join(node["class"]) if isinstance(node["class"], list) else node["class"]
            label.append(f".{classes}")

    # Extract non-empty immediate inline text summary if present
    direct_texts = [
        c.strip() for c in node.children
        if isinstance(c, NavigableString) and not isinstance(c, Comment) and c.strip()
    ]
    if direct_texts:
        text_preview = " ".join(direct_texts)
        if len(text_preview) > max_text_len:
            text_preview = text_preview[:max_text_len] + "..."
        label.append(f'"{text_preview}"')

    current_line = f"{prefix}{connector}{' '.join(label)}"
    lines.append(current_line)

    # Filter direct child element tags
    child_tags = [c for c in node.children if isinstance(c, Tag)]
    total_children = len(child_tags)

    next_prefix = prefix + ("    " if is_last else "│   ")

    for idx, child in enumerate(child_tags):
        child_is_last = (idx == total_children - 1)
        child_lines = render_dom_tree(
            node=child,
            prefix=next_prefix,
            is_last=child_is_last,
            max_depth=max_depth,
            current_depth=current_depth + 1,
            show_attrs=show_attrs,
            max_text_len=max_text_len,
        )
        lines.extend(child_lines)

    return lines

    show_attrs: bool = True,
    output_path: Optional[str] = None,
):
    """Entry point to print or save the tree rendering."""
    target = soup_or_tag.body if hasattr(soup_or_tag, "body") and soup_or_tag.body else soup_or_tag
    tree_lines = render_dom_tree(
        node=target,
        prefix="",
        is_last=True,

  public parseModelResponses(): ModelResponseEntry[] {

   */
  public analyzeResponseNode(responseContainer: HTMLElement): ModelResponseEntry {
    const jslog = responseContainer.getAttribute('jslog');
    const messageContent = responseContainer.querySelector<HTMLElement>(GeminiDomParser.SELECTORS.MESSAGE_CONTENT);
    const footerActions = responseContainer.querySelector<HTMLElement>(GeminiDomParser.SELECTORS.FOOTER_ACTIONS);

    const codeBlocks: CodeBlockDescriptor[] = [];
    if (messageContent) {
      const codeBlockElements = messageContent.querySelectorAll<HTMLElement>(GeminiDomParser.SELECTORS.CODE_BLOCK);
      codeBlockElements.forEach((cbEl) => {
        codeBlocks.push(this.parseCodeBlock(cbEl));
      });
    }

    return {
      root: responseContainer,
      responseContainer,
      jslog,
      messageContent,
      codeBlocks,
      footerActions,
    };
  }

  /**
   * Inspects a code-block element to determine its language, raw content, and header state.
   */
  public parseCodeBlock(codeBlock: HTMLElement): CodeBlockDescriptor {
    const codeElement = codeBlock.querySelector<HTMLElement>('code');
    let language: string | null = null;
    let hasStandardHeader = false;

    // Scan immediate child nodes to find leading TEXT_NODE before the <code> or container element
    for (let i = 0; i < codeBlock.childNodes.length; i++) {
      const child = codeBlock.childNodes[i];
      if (child.nodeType === Node.TEXT_NODE) {
        const textVal = child.textContent?.trim();
        if (textVal && textVal.length > 0) {
          language = textVal;
          hasStandardHeader = true;
          break;
        }
      } else if (child.nodeType === Node.ELEMENT_NODE && (child as HTMLElement).tagName.toLowerCase() === 'code') {
        // Encountered code block before any non-empty text node
        break;
      }
    }

    const normalizedLang = language?.toLowerCase() ?? '';
    const isSnippetOrPlain =
      normalizedLang === 'code snippet' ||
      normalizedLang === 'plain text' ||
      normalizedLang === 'org' ||
      normalizedLang === 'org-mode';

    const rawContent = this.extractRawCode(codeBlock, codeElement);

    return {
      element: codeBlock,
      codeElement,
      language,
      isSnippetOrPlain,
      hasStandardHeader,
      rawContent,
    };
  }

  /**
   * Extracts text payload. If DOM contains styled inner HTML (syntax highlighting tokens),
   * falls back to innerText or copy-buffer interception if available.
   */
  private extractRawCode(codeBlock: HTMLElement, codeElement: HTMLElement | null): string {
    if (!codeElement) {
      return '';
    }

    // Direct content extraction
    // innerText preserves explicit whitespace and line breaks without token span tags
    return codeElement.innerText ?? codeElement.textContent ?? '';
  }

  /**
   * Mounts a custom view replacing or wrapping the target code block.
   */
  public mountDocumentView(
    descriptor: CodeBlockDescriptor,
    viewFactory: (descriptor: CodeBlockDescriptor) => HTMLElement
  ): void {
    const customView = viewFactory(descriptor);
    descriptor.element.style.display = 'none';
    descriptor.element.parentElement?.insertBefore(customView, descriptor.element.nextSibling);
  }

  /**
   * Appends an action node (fold button, copy-as-org) to the response footer actions panel.
   */
  public attachFooterAction(footerElement: HTMLElement, actionButton: HTMLElement): void {
    footerElement.appendChild(actionButton);
  }
}

/**
 * Intercepts the native copy button behavior to retrieve the pure pre-formatted payload.
 */
export async function extractViaCopyButton(codeBlock: HTMLElement): Promise<string | null> {
  const copyBtn = codeBlock.querySelector<HTMLElement>('button[aria-label*="Copy"], .copy-button');
  if (!copyBtn) {
    return null;
  }

  return new Promise((resolve) => {
    const handleCopy = (e: ClipboardEvent) => {
      e.preventDefault();
      e.stopImmediatePropagation();
      const text = e.clipboardData?.getData('text/plain') ?? null;
      document.removeEventListener('copy', handleCopy, true);
      resolve(text);
    };

    document.addEventListener('copy', handleCopy, true);
    copyBtn.click();

    // Fallback timeout to prevent hanging if the click handler does not trigger standard copy event
    setTimeout(() => {
      document.removeEventListener('copy', handleCopy, true);
      resolve(null);
    }, 150);
  });
}

/**
 * Extracts raw text and language identity from a <code-block> element.
 */
export function inspectCodeBlock(codeBlockEl: HTMLElement): ExtractedCodeBlock {
  const codeEl = codeBlockEl.querySelector<HTMLElement>(SELECTORS.CODE);
  
  let language: string | null = null;
  let hasHeader = false;

  // Inspect the first TEXT_NODE child of code-block for language declaration
  const firstChild = codeBlockEl.firstChild;
  if (firstChild && firstChild.nodeType === Node.TEXT_NODE) {
    const text = firstChild.textContent?.trim() || '';
    if (text.length > 0) {
      language = text;
      hasHeader = true;
    }
  }

  // Fallback check if language container is distinct or missing text node
  if (!language && codeEl) {
    // Some variations place language in a data attribute or class
    const langClass = Array.from(codeEl.classList).find(c => c.startsWith('language-'));
    if (langClass) {
      language = langClass.replace('language-', '');
    }
  }

  // Determine if code block contains syntax-highlighted HTML or plain/snippet text
  const isFormattedHtml = codeEl ? codeEl.children.length > 0 : false;

  // Extract raw text content directly
  const rawText = codeEl?.textContent || '';

  return {
    element: codeBlockEl,
    codeElement: codeEl,
    language,
    isFormattedHtml,
    hasHeader,
    rawText,
  };
}

  private observer: MutationObserver | null = null;

    return document.querySelector<HTMLElement>(SELECTORS.CONVERSATION_CONTAINER);

  /**
   * Scans the DOM and mounts custom DocumentViews over target code blocks.
   */
  public mountViews(
    viewRenderer: (block: ExtractedCodeBlock) => HTMLElement | null
  ): void {
    const codeBlocks = this.root.querySelectorAll<HTMLElement>(SELECTORS.CODE_BLOCK);
    
    codeBlocks.forEach(el => {
      // Avoid double-mounting
      if (el.dataset.documentViewMounted === 'true') return;

      const inspected = inspectCodeBlock(el);
      const replacementView = viewRenderer(inspected);

      if (replacementView) {
        el.dataset.documentViewMounted = 'true';
        // Hide original or swap safely
        el.style.display = 'none';
        el.parentNode?.insertBefore(replacementView, el.nextSibling);
      }
    });
  }

  /**
   * Injects actions (such as Copy-As-Org or Fold) into model-response footers.
   */
  public injectFooterAction(
    actionBuilder: (footer: HTMLElement, turn: ModelTurn) => HTMLElement | null
  ): void {
    const modelResponses = this.root.querySelectorAll<HTMLElement>(SELECTORS.MODEL_RESPONSE);

    modelResponses.forEach(el => {
      const turn = parseModelResponse(el);
      if (!turn.footerActions || turn.footerActions.dataset.customActionsMounted === 'true') {
        return;
      }

      const actionButton = actionBuilder(turn.footerActions, turn);
      if (actionButton) {
        turn.footerActions.dataset.customActionsMounted = 'true';
        turn.footerActions.appendChild(actionButton);
      }
    });
  }

  /**
   * Sets up a MutationObserver to react to dynamically streamed turns and code blocks.
   */
  public observe(onNewContent: () => void): void {
    if (this.observer) this.observer.disconnect();

    this.observer = new MutationObserver((mutations) => {
      let shouldTrigger = false;
      for (const mutation of mutations) {
        if (mutation.addedNodes.length > 0) {
          shouldTrigger = true;
          break;
        }
      }
      if (shouldTrigger) {
        onNewContent();
      }
    });

    this.observer.observe(this.root, {
      childList: true,
      subtree: true,
    });
  }

  public disconnect(): void {
    if (this.observer) {
      this.observer.disconnect();
      this.observer = null;
    }
  }

***** Note on Code Block Extraction & Copy Buffer
1. **Extraction without HTML pollution**: The standard `.textContent` property
   on the `<code>` sub-element already resolves HTML entities, strips `<span
   class="...">` tokens, and returns clean plain text. There is rarely a need to
   hook into the copy button's event handler unless custom transforms (like
   stripping Gemini-injected prefixes) are bound via custom clipboard events.
2. **Text Node Language Detection**: Because text nodes can shift across layout
   mutations during streaming, matching `TEXT_NODE` on `code-block` should be
   evaluated once streaming completes or re-evaluated dynamically on each DOM
   mutation.
3. **DocumentView Replacement Strategy**: Replacing in-place can break
   Angular/Polymer/Lit event listeners or cause re-render loops if Gemini
   re-evaluates the virtual node. The safest pattern is to set
   `codeBlockEl.style.display = 'none'` and insert the custom `DocumentView`
   adjacent to it (`insertBefore(customView, codeBlockEl.nextSibling)`).

**** Architectural Overview (MVC Separation)
To keep clean boundaries within the webextension content script context:
- *Model*: Domain entities representing parsed conversational structures (`Conversation`, `Message`, `UserQuery`, `ModelResponse`, `CodeBlockTarget`).
- *View*: Preact 11 root mounts (`DocumentView` wrappers, message action bar extensions).

**** Implementation Strategy & Edge Case Mitigations
***** 1. Plaintext Extraction vs Highlighted Syntax
- Instead of dealing with formatted syntax HTML or clipboard events directly, grab `element.textContent` on the `<code>` node, or leverage `navigator.clipboard` interception if raw carriage returns are sanitized by Gemini's internal renderer
.- For language parsing, inspect the header node's `childNodes` for the leading text node before the button container. Move on to language detection if it reports ~Code snippet~ or ~Plain text~ to identify languages we support that the chat does not.
***** 2. Streaming DOM & Debounced Injection
- Model responses stream incrementally. Do not attempt full AST parses on every tiny text mutation.
- Use a `MutationObserver` targeting `infinite-scroller` / `.conversation-container` with `childList: true, subtree: true`.
- Check `dataset` flags (e.g. `data-ext-processed="true"`) to prevent re-parsing completed blocks and avoid layout thrashing.

import { ParsedCodeBlock } from '../types';
import { SELECTORS } from '../selectors';

export function parseCodeBlock(codeBlockEl: HTMLElement): ParsedCodeBlock | null {
  const codeEl = codeBlockEl.querySelector<HTMLElement>(SELECTORS.CODE_ELEMENT);
  if (!codeEl) return null;

  // Language resolution
  let language = 'text';
  let hasCustomHeader = false;

  // Find first non-empty text node in code-block (outside the code tag)
  const walker = document.createTreeWalker(
    codeBlockEl,
    NodeFilter.SHOW_TEXT,
    {
      acceptNode: (node) => {
        if (codeEl.contains(node)) return NodeFilter.FILTER_REJECT;
        return node.textContent?.trim() ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_SKIP;
      }
    }
  );

  const headerTextNode = walker.nextNode();
  if (headerTextNode && headerTextNode.textContent) {
    const rawLang = headerTextNode.textContent.trim().toLowerCase();
    hasCustomHeader = true;
    if (rawLang === 'code snippet' || rawLang === 'plain text') {
      language = 'text';
    } else {
      language = rawLang;
    }
  }

  // Generate or read stable identifier
  if (!codeBlockEl.dataset.extBlockId) {
    codeBlockEl.dataset.extBlockId = `cb-${crypto.randomUUID()}`;
  }

  return {
    id: codeBlockEl.dataset.extBlockId,
    element: codeBlockEl,
    codeElement: codeEl,
    language,
    rawText: codeEl.textContent ?? '',
    hasCustomHeader
  };
}

export class PreactMountManager {
  private mountedRoots = new Map<string, HTMLElement>();

  public mountDocumentView(block: ParsedCodeBlock, viewComponent: preact.ComponentType<{ content: string; language: string }>): void {
    if (block.element.dataset.extMounted === 'true') return;

    // Check if this matches org-mode format (starts with Org headline / stamp or declared as org)
    const isOrg = block.language === 'org' || block.rawText.trimStart().startsWith('* ');

    block.element.style.display = 'none';
    block.element.parentNode?.insertBefore(mountContainer, block.element.nextSibling);

    render(h(viewComponent, { content: block.rawText, language: block.language }), mountContainer);

    block.element.dataset.extMounted = 'true';
    this.mountedRoots.set(block.id, mountContainer);
  }

  public mountFooterActions(msg: ParsedModelMessage, actionsComponent: preact.ComponentType<{ message: ParsedModelMessage }>): void {
    if (!msg.footerElement || msg.footerElement.dataset.extActionsMounted === 'true') return;

    const container = document.createElement('div');
    container.className = 'ext-message-actions-slot';
    msg.footerElement.appendChild(container);

    render(h(actionsComponent, { message: msg }), container);
    msg.footerElement.dataset.extActionsMounted = 'true';
  }

`domObserver.ts` (Controller): Owns the raw DOM mutation lifecycle, debouncing, and event emission. It should have no knowledge of specific views or Preact injection.
#+begin_src typescript
import { SELECTORS } from '../selectors';

export interface MutationEvents {
  onNodesAdded?: () => void;
}

export class GeminiDomObserver {
  private observer: MutationObserver | null = null;
  private debounceTimer: number | null = null;

  constructor(private callbacks: MutationEvents, private debounceMs = 50) {}

  public observe(targetRoot: ParentNode = document): void {
    const target = targetRoot.querySelector(SELECTORS.CONVERSATION_CONTAINER) || document.body;

    this.observer = new MutationObserver((mutations) => {
      let hasAdditions = false;
      for (const mutation of mutations) {
        if (mutation.addedNodes.length > 0) {
          hasAdditions = true;
          break;
        }
      }

      if (hasAdditions) {
        this.scheduleNotification();
      }
    });

    this.observer.observe(target, {
      childList: true,
      subtree: true,
    });
  }

  private scheduleNotification(): void {
    if (this.debounceTimer !== null) {
      window.clearTimeout(this.debounceTimer);
    }
    this.debounceTimer = window.setTimeout(() => {
      this.callbacks.onNodesAdded?.();
      this.debounceTimer = null;
    }, this.debounceMs);
  }

  public disconnect(): void {
    if (this.debounceTimer !== null) {
      window.clearTimeout(this.debounceTimer);
      this.debounceTimer = null;
    }
    this.observer?.disconnect();
    this.observer = null;
  }
}

** Design Tradeoffs & Benefits
1. *Zero Site Leaks in Core Views*: `OrgDocumentView` only expects an AST or raw
   string content; it has no dependencies on `HTMLElement` nodes from Google or
   Anthropic.
2. *Isolated Blast Radius*: When Gemini changes its DOM class names or custom
   elements, only `src/chat/gemini/` requires updates. The rest of the pipeline
   remains untouched.
3. *Fast Hot-Reloading & Testing*: Parsing logic in `src/core/ast/` and
   rendering logic in `src/views/` can be tested via standard headless unit
   tests (e.g. Vitest/Jest) without mocking the Gemini DOM.
